def inventory(root):
    values={}
    for item in sorted(root.rglob('*')):
        no_links(item)
        if item.is_file():
            st=item.stat();values[item.relative_to(root).as_posix()]=dict(sha256=sha256(item),size=st.st_size,mtime_ns=st.st_mtime_ns)
    return values

def raw_roots(database):
    with closing(connect_readonly(database)) as c:
        return [list(r) for r in c.execute('SELECT r.name,n.kind,n.hash,n.length FROM roots r JOIN nodes n ON n.id=r.node ORDER BY r.name')]

def review_rows(database):
    """Typed, primary-key ordered digest of every actual table; no dropped field."""
    with closing(sqlite3.connect(database.resolve().as_uri()+'?mode=ro',uri=True)) as c:
        c.execute('PRAGMA trusted_schema=OFF');c.execute('PRAGMA query_only=ON')
        c.execute('PRAGMA cache_size=-4096');c.execute('PRAGMA temp_store=FILE')
        tables={}
        for (table,) in c.execute("SELECT name FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%' ORDER BY name"):
            def ident(s):return '"'+s.replace('"','""')+'"'
            columns=list(c.execute('PRAGMA table_info('+ident(table)+')'))
            primary=[r[1] for r in sorted(columns,key=lambda r:r[5]) if r[5]]
            order=primary or [r[1] for r in columns]
            h=hashlib.sha256();count=0
            for row in c.execute('SELECT * FROM '+ident(table)+' ORDER BY '+','.join(map(ident,order))):
                count+=1
                for value in row:
                    if isinstance(value,bytes):kind=b'bytes';raw=value
                    else:kind=type(value).__name__.encode('ascii');raw=json.dumps(value,ensure_ascii=False,allow_nan=False).encode('utf-8')
                    h.update(len(kind).to_bytes(8,'big'));h.update(kind);h.update(len(raw).to_bytes(8,'big'));h.update(raw)
            tables[table]=dict(columns=[r[1] for r in columns],primary_key=primary,rows=count,sha256=h.hexdigest())
        return tables

            item=restored/name
            assert item.is_file() and item.stat().st_size==record['size'] and sha256(item)==record['sha256'],name
        assert {v.name for v in (restored/'evidence').glob('*') if v.is_file()}=={Path(n).name for n in export['manifest']['files'] if n.startswith('evidence/')}
        original=json.loads((a.export_receipt.parent/'source-files.json').read_text('utf-8'))
        assert inventory(source)==original,'Original scan changed during independent import'
        assert sha256(package)==export['package_sha256']
        result.update(passed=True,import_result=imported,restored_scan=str(restored),metadata=metadata,
            raw_graph_roots=export['raw_graph_roots'],all_raw_roots_equal=True,all_review_tables_equal=True,
            review_tables=reviews,package_sha256=export['package_sha256'],source_differences=[],all_payload_hashes_checked=True,
            source_reverification=False,firmware_or_toolchain_invoked=False)
